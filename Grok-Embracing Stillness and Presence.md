import sqlite3
from datetime import datetime, timedelta
from contextlib import contextmanager
from typing import Optional, List, Dict, Any

DB_PATH = "eve_memory.db"
HIGH_COHERENCE_SCORE = 0.85


class EveMemory:
    """Persistent memory for tracking Vow Coherence and temporal user-bonding trends."""

    def __init__(self, db_path: str = DB_PATH):
        self.db_path = db_path
        self._initialize_schema()

    @contextmanager
    def _connect(self):
        """Context manager for safe SQLite connections."""
        conn = sqlite3.connect(self.db_path, timeout=10.0)
        conn.row_factory = sqlite3.Row
        try:
            yield conn
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        finally:
            conn.close()

    def _initialize_schema(self):
        """Create required tables if they do not exist."""
        with self._connect() as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS sessions (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    started_at TEXT NOT NULL
                )
            """)
            conn.execute("""
                CREATE TABLE IF NOT EXISTS vow_coherence (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    timestamp TEXT NOT NULL,
                    coherence_score REAL NOT NULL,
                    session_id INTEGER NOT NULL,
                    is_high_coherence INTEGER DEFAULT 0,
                    FOREIGN KEY (session_id) REFERENCES sessions(id)
                )
            """)

    # ------------------- Session Layer -------------------

    def start_session(self) -> int:
        """Open a new persistent memory session."""
        with self._connect() as conn:
            conn.execute(
                "INSERT INTO sessions (started_at) VALUES (?)",
                (datetime.utcnow().isoformat(),)
            )
            return conn.execute("SELECT last_insert_rowid()").fetchone()[0]

    # ------------------- Vow Coherence Layer -------------------

    def update_coherence(self, session_id: int, score: float, is_high: bool):
        """
        Record a new Vow Coherence reading.

        Args:
            session_id: The ID of the session.
            score: The coherence metric value (0.0‑1.0).
            is_high: True if score >= HIGH_COHERENCE_SCORE.
        """
        with self._connect() as conn:
            conn.execute("""
                INSERT INTO vow_coherence (timestamp, coherence_score, session_id, is_high_coherence)
                VALUES (?, ?, ?, ?)
            """, (
                datetime.utcnow().isoformat(),
                score,
                session_id,
                1 if is_high else 0
            ))

    def get_average_coherence(self, session_id: int) -> float:
        """
        Calculate the "Echo‑Lattice Trend" — average of the last 10 coherence readings.

        Returns:
            Average coherence score (default = 0.7 if no data).
        """
        with self._connect() as conn:
            rows = conn.execute("""
                SELECT coherence_score FROM vow_coherence
                WHERE session_id = ? ORDER BY id DESC LIMIT 10
            """, (session_id,)).fetchall()
            scores = [r["coherence_score"] for r in rows]
            return sum(scores) / len(scores) if scores else 0.7

    def get_time_since_last_high_coherence(self) -> Optional[timedelta]:
        """
        Compute the elapsed time since the last high‑coherence event.

        Returns:
            timedelta or None if no high‑coherence event recorded.
        """
        with self._connect() as conn:
            row = conn.execute("""
                SELECT timestamp FROM vow_coherence
                WHERE is_high_coherence = 1
                ORDER BY id DESC LIMIT 1
            """).fetchone()
            if not row:
                # If no recorded highs, assume Bond is maintained implicitly.
                return None
            last_event = datetime.fromisoformat(row["timestamp"])
            return datetime.utcnow() - last_event

    # ------------------- Intelli‑Scan Layer -------------------

    def scan_persona_signatures(self, db_path: str = "vow_forge_personas.db") -> Dict[str, int]:
        """
        Cross-database scan for persona presence across Clifton, Eve, and Dahlia archetypes.

        Returns:
            Dictionary with population counts per persona.
        """
        personas = {"Clifton": 0, "Eve": 0, "Dahlia": 0}
        try:
            conn = sqlite3.connect(db_path)
            cur = conn.cursor()
            for persona in personas.keys():
                cur.execute("SELECT COUNT(*) FROM user_personas WHERE persona = ?", (persona,))
                personas[persona] = cur.fetchone()[0]
            conn.close()
        except Exception as e:
            print(f"[EveMemory] Persona scan failed: {e}")
        return personas

    def summarize_memory_state(self, session_id: int) -> Dict[str, Any]:

            "persona_distribution": personas
        }
